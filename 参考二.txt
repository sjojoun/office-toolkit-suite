function dedupLines(text, sort, dir){
  var lines = [...new Set(text.split('\n').filter(function(l){ return l.trim(); }))];
  if (sort) lines.sort(function(a,b){ return dir===1?a.localeCompare(b):b.localeCompare(a); });
  return lines.join('\n');
}

function runConversion(){
  var text = convInput.value;
  if (!text.trim()) { toast('请先输入内容','warning'); return; }
  if (checkVipText(text)) return;
  var result = '';
  try {
    switch(convType){
      case 'base64': {
        var isEnc = !$('#b64Dec') || !$('#b64Dec').classList.contains('active');
        result = isEnc ? btoa(unescape(encodeURIComponent(text))) : decodeURIComponent(escape(atob(text.trim())));
        break;
      }
      case 'url': {
        var isEnc2 = !$('#urlDec') || !$('#urlDec').classList.contains('active');
        result = isEnc2 ? encodeURIComponent(text) : decodeURIComponent(text);
        break;
      }
      case 'json':
        result = JSON.stringify(JSON.parse(text), null, 2);
        break;
      case 'xml': {
        var isX2j = !$('#j2xBtn') || !$('#j2xBtn').classList.contains('active');
        result = isX2j ? xmlToJson(text) : jsonToXml(text);
        break;
      }
      case 'csv': {
        var isC2j = !$('#j2cBtn') || !$('#j2cBtn').classList.contains('active');
        result = isC2j ? csvToJson(text) : jsonToCsv(text);
        break;
      }
      case 'timestamp': {
        var isTsMode = !$('#d2tsBtn') || !$('#d2tsBtn').classList.contains('active');
        if (isTsMode) {
          var ts = parseInt(text.trim(), 10);
          if (isNaN(ts)) { toast('请输入有效的时间戳','error'); return; }
          var ms = ts > 9999999999 ? ts : ts * 1000;
          var d = new Date(ms);
          result = 'UTC: '+d.toUTCString()+'\n本地: '+d.toLocaleString()+'\nISO: '+d.toISOString()+'\n时间戳(ms): '+d.getTime()+'\n时间戳(s): '+Math.floor(d.getTime()/1000);
        } else {
          var d2 = new Date(text.trim());
          if (isNaN(d2.getTime())) { toast('请输入有效的日期字符串','error'); return; }
          result = '时间戳(秒): '+Math.floor(d2.getTime()/1000)+'\n时间戳(毫秒): '+d2.getTime()+'\nISO: '+d2.toISOString()+'\n本地: '+d2.toLocaleString();
        }
        break;
      }
      case 'case': case 'dedup':
        toast('请点击上方具体操作按钮','warning'); return;
    }
    if (result === undefined || result === null) result = '';
    convOutput.value = String(result);
    updateConvStats();
  } catch(e) {
    toast('操作失败: '+e.message,'error');
  }
}

/* ─── XML helpers ─── */
function xmlToJson(xmlStr){
  try{
    var parser=new DOMParser();
    var doc=parser.parseFromString(xmlStr,'text/xml');
    if(doc.querySelector('parsererror')){ toast('XML 解析错误','error'); return ''; }
    return JSON.stringify(xmlNodeToObj(doc.documentElement),null,2);
  }catch(e){ toast('XML 转换失败: '+e.message,'error'); return ''; }
}
function xmlNodeToObj(node){
  var obj={};
  for(var i=0;i<node.attributes.length;i++) obj['@'+node.attributes[i].name]=node.attributes[i].value;
  var children=Array.from(node.children);
  if(!children.length){
    var t=node.textContent.trim();
    if(t){ if(!Object.keys(obj).length) return t; obj['#text']=t; }
    return Object.keys(obj).length?obj:'';
  }
  children.forEach(function(child){
    var tag=child.tagName, val=xmlNodeToObj(child);
    if(obj[tag]===undefined) obj[tag]=val;
    else{ if(!Array.isArray(obj[tag])) obj[tag]=[obj[tag]]; obj[tag].push(val); }
  });
  return obj;
}
function jsonToXml(jsonStr){
  try{
    var obj=JSON.parse(jsonStr);
    return '<?xml version="1.0" encoding="UTF-8"?>\n'+formatXml(objToXml(obj,'root'));
  }catch(e){ toast('JSON 解析失败: '+e.message,'error'); return ''; }
}
function objToXml(obj,tag){
  if(typeof obj!=='object'||obj===null) return '<'+tag+'>'+escXml(String(obj))+'</'+tag+'>';
  if(Array.isArray(obj)) return obj.map(function(item){ return objToXml(item,tag); }).join('');
  var xml='<'+tag+'>';
  for(var k in obj){ if(Object.prototype.hasOwnProperty.call(obj,k)&&k[0]!=='@') xml+=objToXml(obj[k],k); }
  return xml+'</'+tag+'>';
}
function escXml(s){ return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }
function formatXml(xml){
  var f='',ind=0,lines=xml.replace(/>\s*</g,'>\n<').split('\n');
  lines.forEach(function(line){
    line=line.trim();
    if(!line)return;
    if(line.match(/^<\/\w/)) ind=Math.max(0,ind-1);
    f+='  '.repeat(ind)+line+'\n';
    if(line.match(/^<\w[^>]*[^\/]>$/)&&!line.match(/^<\?/)) ind++;
  });
  return f.trim();
}

/* ─── CSV helpers ─── */
function csvToJson(csv){
  try{
    var lines=csv.trim().split('\n');
    if(lines.length<2){ toast('CSV 至少需要表头+一行数据','error'); return ''; }
    var hdrs=parseCsvLine(lines[0]);
    var result=[];
    for(var i=1;i<lines.length;i++){
      if(!lines[i].trim()) continue;
      var vals=parseCsvLine(lines[i]);
      var row={};
      hdrs.forEach(function(h,idx){ row[h]=vals[idx]||''; });
      result.push(row);
    }
    return JSON.stringify(result,null,2);
  }catch(e){ toast('CSV 转换失败: '+e.message,'error'); return ''; }
}
function parseCsvLine(line){
  var res=[],cur='',inQ=false;
  for(var i=0;i<line.length;i++){
    var ch=line[i];
    if(inQ){ if(ch==='"'){ if(line[i+1]==='"'){cur+='"';i++;} else inQ=false; } else cur+=ch; }
    else{ if(ch==='"') inQ=true; else if(ch===','){res.push(cur.trim());cur='';} else cur+=ch; }
  }
  res.push(cur.trim());
  return res;
}
function jsonToCsv(json){
  try{
    var data=JSON.parse(json);
    var arr=Array.isArray(data)?data:[data];
    if(!arr.length){ toast('JSON 数组为空','warning'); return ''; }
    var hdrs=Object.keys(arr[0]);
    var lines=[hdrs.join(',')];
    arr.forEach(function(row){
      lines.push(hdrs.map(function(h){
        var v=String(row[h]!==undefined?row[h]:'');
        return (v.indexOf(',')>-1||v.indexOf('"')>-1)?'"'+v.replace(/"/g,'""')+'"':v;
      }).join(','));
    });
    return lines.join('\n');
  }catch(e){ toast('JSON 解析失败: '+e.message,'error'); return ''; }
}

function renderImg(){
  if(!origImg) return;
  var sw = origImg.naturalWidth || origImg.width;
  var sh = origImg.naturalHeight || origImg.height;
  var rw = imgRot%180!==0 ? sh : sw;
  var rh = imgRot%180!==0 ? sw : sh;
  var maxW = canvasWrap.offsetWidth - 32 || 800;
  var maxH = 520;
  var scale = Math.min(maxW/rw, maxH/rh, 1);
  var cw = Math.round(rw*scale);
  var ch = Math.round(rh*scale);
  if(imgCanvas.width!==cw) imgCanvas.width=cw;
  if(imgCanvas.height!==ch) imgCanvas.height=ch;
  ctx.clearRect(0,0,cw,ch);
  ctx.save();
  var b=$('#brightSlider').value, c=$('#contSlider').value, s=$('#satSlider').value;
  var f='';
  if(b!=100) f+='brightness('+b+'%) ';
  if(c!=100) f+='contrast('+c+'%) ';
  if(s!=100) f+='saturate('+s+'%) ';
  if(f) ctx.filter=f.trim();
  ctx.translate(cw/2,ch/2);
  if(imgFlipH) ctx.scale(-1,1);
  if(imgFlipV) ctx.scale(1,-1);
  ctx.rotate(imgRot*Math.PI/180);
  ctx.drawImage(origImg, -sw*scale/2, -sh*scale/2, sw*scale, sh*scale);
  ctx.restore();
}

function runRegex(){
  var text=regexInput.value, pat=regexPat.value.trim(), flags=regexFlags.value.trim();
  regexErr.style.display='none'; regexStats.style.display='none';
  if(!pat){ regexResult.innerHTML='<span style="color:var(--text-3)">等待输入正则表达式…</span>'; return; }
  if(!text){ regexResult.innerHTML='<span style="color:var(--text-3)">请输入测试文本</span>'; return; }
  if(text.length>VIP_TEXT){ showVip(); return; }
  try{
    var rx=new RegExp(pat, flags||'g');
    var matches=[], m;
    var safePat=new RegExp(pat, flags.indexOf('g')>=0?flags:(flags+'g'));
    while((m=safePat.exec(text))!==null){
      matches.push({start:m.index, len:m[0].length, text:m[0]});
      if(!safePat.global && !safePat.sticky) break;
      if(m[0].length===0) safePat.lastIndex++;
      if(matches.length>5000) break;
    }
    if(!matches.length){
      regexResult.innerHTML='<span style="color:var(--text-3)">无匹配结果</span>';
    }else{
      var h='', last=0;
      matches.forEach(function(hi){
        h+=escHtml(text.slice(last,hi.start));
        h+='<span class="match-hl">'+escHtml(hi.text)+'</span>';
        last=hi.start+hi.len;
      });
      h+=escHtml(text.slice(last));
      regexResult.innerHTML=h;
      $('#matchCount').textContent=matches.length;
      regexStats.style.display='flex';
    }
  }catch(e){
    regexErr.textContent='⚠ '+e.message; regexErr.style.display='block';
    regexResult.innerHTML='<span style="color:var(--text-3)">正则表达式有误</span>';
  }
}
function escHtml(s){ var d=document.createElement('div'); d.textContent=s; return d.innerHTML; }

function formatJS(code,indent){
  var sp=''.padStart(indent,' '), level=0, result='', i=0;
  var inStr=false, strCh='', inBlockC=false, inLineC=false;
  while(i<code.length){
    var ch=code[i], prev=i>0?code[i-1]:'';
    if(!inBlockC&&!inLineC&&(ch==='"'||ch==="'"||ch==='`')){
      if(!inStr){inStr=true;strCh=ch;}
      else if(ch===strCh&&prev!=='\\'){ inStr=false; }
    }
    if(!inStr&&!inBlockC&&!inLineC&&ch==='/'&&code[i+1]==='*'){inBlockC=true;}
    if(inBlockC&&ch==='*'&&code[i+1]==='/'){ result+='*/'; i+=2; inBlockC=false; continue; }
    if(!inStr&&!inBlockC&&!inLineC&&ch==='/'&&code[i+1]==='/'){ inLineC=true; }
    if(inLineC&&ch==='\n'){ inLineC=false; result+='\n'; i++; continue; }
    if(!inStr&&!inBlockC&&!inLineC){
      if(ch==='{'){result+=ch+'\n'; level++; result+=sp.repeat(level); i++; continue;}
      if(ch==='}'){level=Math.max(0,level-1); result=result.trimEnd()+'\n'+sp.repeat(level)+'}'; i++; continue;}
      if(ch===';'){result+=ch+'\n'+sp.repeat(level); i++; continue;}
    }
    result+=ch; i++;
  }
  return result;
}
function formatCSS(code,indent){
  var sp=''.padStart(indent,' '), result='', lines=code.replace(/\{/g,'{\n').replace(/\}/g,'\n}\n').replace(/;/g,';\n').split('\n');
  var lv=0;
  lines.forEach(function(line){
    line=line.trim();
    if(!line)return;
    if(line==='}'){lv=Math.max(0,lv-1); result+=sp.repeat(lv)+line+'\n';}
    else if(line.endsWith('{')){ result+=sp.repeat(lv)+line+'\n'; lv++; }
    else{ result+=sp.repeat(lv)+line+'\n'; }
  });
  return result.trim();
}
function formatHTML(code,indent){
  var sp=''.padStart(indent,' '), lv=0, result='';
  var tokens=code.replace(/>\s*</g,'>\n<').split('\n');
  tokens.forEach(function(t){
    t=t.trim();
    if(!t)return;
    if(t.match(/^<\//)) lv=Math.max(0,lv-1);
    result+=sp.repeat(lv)+t+'\n';
    if(t.match(/^<[^\/!?][^>]*[^\/]>$/)&&!t.match(/^<(area|base|br|col|embed|hr|img|input|link|meta|param|source|track|wbr)/i)) lv++;
  });
  return result.trim();
}

  batchProg.style.display='block';

    }
  }
  requestAnimationFrame(tick);
}

  } else { showConvErr('请先上传文件或输入文本'); }
}

/* PDF → plain text */
function convPdfToText(ab){
  if(typeof pdfjsLib==='undefined'){ showConvErr('pdf.js 加载失败'); return; }
  docProgUpdate(20,'正在解析 PDF…');
  pdfjsLib.getDocument({data:ab}).promise.then(function(pdf){
    var total=pdf.numPages, done=0, fullText='';
    function doPage(n){
      pdf.getPage(n).then(function(page){
        return page.getTextContent();
      }).then(function(tc){
        var pt='', lastY=null;
        tc.items.forEach(function(item){
          if(lastY!==null && Math.abs(item.transform[5]-lastY)>4) pt+='\n';
          else if(pt&&lastY!==null) pt+=' ';
          pt+=item.str;
          lastY=item.transform[5];
        });
        fullText+='\n── 第 '+n+' 页 ──\n'+pt;
        done++;
        docProgUpdate(20+Math.round(done/total*70),'提取文字 ('+done+'/'+total+')…');
        if(done>=total){
          docProgUpdate(100,'提取完成！');
          setTimeout(function(){
            $('#docProg').classList.remove('show');
            $('#docResult').classList.add('show');
            $('#docResultTxt').textContent=fullText;
            toast('已提取 '+total+' 页文字');
          },300);
        }
      }).catch(function(e){ showConvErr('页面 '+n+' 解析失败: '+e.message); });
    }
    for(var i=1;i<=total;i++) doPage(i);
  }).catch(function(e){ showConvErr('PDF 解析失败: '+e.message); });
}

/* ── PDF → Word: DEEP REWRITE WITH HEURISTIC LAYOUT RESTORATION ── */
function convPdfToWord(ab){
  if(typeof pdfjsLib==='undefined'){ showConvErr('pdf.js 加载失败'); return; }
  if(typeof docx==='undefined'){     showConvErr('docx 库加载失败'); return; }
  docProgUpdate(10,'正在解析 PDF 结构…');

  pdfjsLib.getDocument({data:ab}).promise.then(function(pdf){
    var total=pdf.numPages, done=0, allParas=[];

    function doPage(pn){
      pdf.getPage(pn).then(function(page){
        var vp=page.getViewport({scale:1.0});
        return page.getTextContent({includeMarkedContent:false}).then(function(tc){
          return {tc:tc, vw:vp.width, vh:vp.height};
        });
      }).then(function(r){
        var tc=r.tc, vw=r.vw;

        /* Step 1: collect items with coordinates & font size */
        var items=[];
        tc.items.forEach(function(item){
          if(item.type==='beginMarkedContent'||item.type==='endMarkedContent') return;
          var str=item.str||'';
          var x=item.transform[4];
          var y=item.transform[5];
          /* Font size from transform matrix: sqrt(a^2+b^2) */
          var fs=Math.sqrt(item.transform[0]*item.transform[0]+item.transform[1]*item.transform[1]);
          fs=Math.max(fs,6);
          items.push({str:str,x:x,y:y,w:item.width||0,fs:fs});
        });
        if(!items.length){ done++; check(); return; }

        /* Step 2: cluster into lines by Y proximity (tolerance = 2pt) */
        items.sort(function(a,b){
          var dy=Math.abs(a.y-b.y);
          if(dy<2) return a.x-b.x;
          return b.y-a.y;
        });
        var lines=[], cur=null;
        var LINE_TOL=2;
        items.forEach(function(item){
          if(!cur){ cur={y:item.y,items:[item],fs:item.fs}; }
          else if(Math.abs(item.y-cur.y)<=LINE_TOL){
            cur.items.push(item);
            cur.y=(cur.y*(cur.items.length-1)+item.y)/cur.items.length;
            cur.fs=Math.max(cur.fs,item.fs);
          }else{
            lines.push(cur);
            cur={y:item.y,items:[item],fs:item.fs};
          }
        });
        if(cur) lines.push(cur);

        /* Step 3: reconstruct line text; detect indent */
        var pageLeft=vw;
        lines.forEach(function(line){
          line.items.sort(function(a,b){ return a.x-b.x; });
          if(line.items.length) pageLeft=Math.min(pageLeft,line.items[0].x);
          var txt='', prev=null;
          line.items.forEach(function(item){
            if(!prev){ txt=item.str; }
            else{
              var gap=item.x-(prev.x+prev.w);
              if(gap > prev.fs*0.30) txt+=' ';
              txt+=item.str;
            }
            prev=item;
          });
          line.text=txt.trim();
          line.indent=line.items.length?line.items[0].x:0;
        });
        /* Remove purely empty lines */
        lines=lines.filter(function(l){ return l.text.length>0; });

        /* Step 4: compute avg line height & avg font size */
        var avgFs=lines.reduce(function(s,l){ return s+l.fs; },0)/Math.max(lines.length,1);
        var lhArr=[];
        for(var i=1;i<lines.length;i++) lhArr.push(Math.abs(lines[i-1].y-lines[i].y));
        var avgLh=lhArr.length ? lhArr.reduce(function(a,b){return a+b;},0)/lhArr.length : 14;

        /* Step 5: page break separator (except page 1) */
        if(pn>1){
          allParas.push(new docx.Paragraph({
            children:[new docx.TextRun({text:'',break:1})],
            spacing:{before:240,after:240}
          }));
        }

        /* Step 6: build docx paragraphs */
        for(var i=0;i<lines.length;i++){
          var line=lines[i];
          var prevY=i>0?(lines[i-1].y-line.y):avgLh;
          var isHeading = line.fs > avgFs*1.25 && line.text.length<120;
          var isNewPara = prevY > avgLh*1.55;
          var isIndented= line.indent > pageLeft+8;

          /* Font size in half-points (Word unit) */
          var halfPt=Math.round(Math.max(line.fs,8)*2);
          var runOpts={text:line.text, size:halfPt};
          if(isHeading) runOpts.bold=true;

          var paraOpts={
            children:[new docx.TextRun(runOpts)],
            spacing:{
              before: isHeading?300:(isNewPara?180:0),
              after:  isHeading?180:80,
              line:360
            }
          };
          if(isIndented){
            var indentTwips=Math.round((line.indent-pageLeft)*20);
            paraOpts.indent={left:Math.min(indentTwips,2880)};
          }
          if(isHeading){
            /* Use Heading 2 style for detected headings */
            try{ paraOpts.heading=docx.HeadingLevel.HEADING_2; }catch(ex){}
          }
          allParas.push(new docx.Paragraph(paraOpts));
        }

        done++;
        docProgUpdate(10+Math.round(done/total*70),'处理中 ('+done+'/'+total+' 页)…');
        check();
      }).catch(function(e){ showConvErr('页面 '+pn+' 解析失败: '+e.message); });
    }

    function check(){
      if(done>=total){
        docProgUpdate(85,'正在生成 Word 文档…');
        buildWordDoc(allParas);
      }
    }

    for(var i=1;i<=total;i++) doPage(i);
  }).catch(function(e){ showConvErr('PDF 解析失败: '+e.message); });
}

function buildWordDoc(paragraphs){
  try{
    var doc=new docx.Document({
      creator:'Office Toolkit Suite',
      title:'Converted Document',
      styles:{
        default:{
          document:{run:{font:'Arial',size:22}}
        }
      },
      sections:[{
        properties:{
          page:{margin:{top:1440,right:1080,bottom:1440,left:1080}}
        },
        children:paragraphs.length ? paragraphs : [
          new docx.Paragraph({children:[new docx.TextRun({text:'（文档内容为空或无法提取）'})]})
        ]
      }]
    });
    docx.Packer.toBlob(doc).then(function(blob){
      var name=(docFile?docFile.name:'document').replace(/\.pdf$/i,'.docx');
      finishConv(blob,name,'application/vnd.openxmlformats-officedocument.wordprocessingml.document');
    }).catch(function(e){ showConvErr('Word 生成失败: '+e.message); });
  }catch(e){ showConvErr('Word 生成失败: '+e.message); }
}

/* Plain text → Word */
function convTextToWord(text){
  if(typeof docx==='undefined'){ showConvErr('docx 库加载失败'); return; }
  docProgUpdate(40,'正在生成 Word 文档…');
  setTimeout(function(){
    try{
      var paras=text.split('\n').map(function(line){
        return new docx.Paragraph({
          children:[new docx.TextRun({text:line||' ',size:22})],
          spacing:{after:120}
        });
      });
      var doc=new docx.Document({sections:[{properties:{},children:paras}]});
      docx.Packer.toBlob(doc).then(function(blob){
        finishConv(blob,'converted.docx','application/vnd.openxmlformats-officedocument.wordprocessingml.document');
      }).catch(function(e){ showConvErr('Word 生成失败: '+e.message); });
    }catch(e){ showConvErr('Word 生成失败: '+e.message); }
  },80);
}

/* Plain text → PDF (via browser print) */
function convTextToPdf(text){
  docProgUpdate(30,'正在准备打印内容…');
  var paras=text.split('\n').map(function(l){
    return '<p style="margin:0 0 4px">'+(l||'&nbsp;')+'</p>';
  }).join('');
  var bodyHtml='<div style="font-family:\'Microsoft YaHei\',\'PingFang SC\',sans-serif;font-size:11pt;color:#000;">'+paras+'</div>';
  printToPdf(bodyHtml,'converted');
}

function printToPdf(bodyHtml,title){
  docProgUpdate(70,'正在打开打印窗口…');
  var pw=window.open('','_blank','width=900,height=700');
  if(!pw){ showConvErr('弹窗被拦截，请允许弹出窗口后重试'); return; }
  var fullHtml='<!DOCTYPE html><html><head><meta charset="UTF-8"><title>'+title+'</title>'+
    '<style>@media print{@page{size:A4;margin:15mm 12mm;}body{-webkit-print-color-adjust:exact;print-color-adjust:exact;}}'+
    'body{font-family:"Microsoft YaHei","PingFang SC","Noto Sans SC",sans-serif;font-size:11pt;line-height:1.8;color:#000;padding:20px;}'+
    'table{border-collapse:collapse;width:100%;}td,th{border:1px solid #ccc;padding:4px 8px;}img{max-width:100%;}</style>'+
    '</head><body>'+bodyHtml+'</body></html>';
  var blob=new Blob([fullHtml],{type:'text/html;charset=UTF-8'});
  var burl=URL.createObjectURL(blob);
  pw.onload=function(){
    URL.revokeObjectURL(burl);
    docProgUpdate(90,'请在打印对话框中选择"另存为 PDF"…');
    setTimeout(function(){
      pw.print();
      pw.onafterprint=function(){ pw.close(); resetDocResult(); };
    },500);
  };
  pw.location.href=burl;
  setTimeout(function(){ resetDocResult(); },8000);
}
